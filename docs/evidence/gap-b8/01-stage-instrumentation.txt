static const char *const sm750_mode_stage_names[] = {
	[SM750_MODE_STAGE_NONE] = "none",
	[SM750_MODE_STAGE_SII_BEGIN] = "sii902x_begin_mode",
	[SM750_MODE_STAGE_DISPLAY_CTRL] = "ddk750_set_display_control",
	[SM750_MODE_STAGE_TIMING] = "ddk750_set_mode_timing",
	[SM750_MODE_STAGE_LOGICAL_OUT] = "ddk750_set_logical_disp_out",
	[SM750_MODE_STAGE_SII_ENABLE] = "sii902x_enable",
};

static const char *sm750_mode_stage_name(enum sm750_mode_stage stage)
{
	if ((unsigned int)stage >= ARRAY_SIZE(sm750_mode_stage_names))
		return "unknown";
	return sm750_mode_stage_names[stage];
}

--- 7. the policy gate added for this branch (tests/check-drm-policy.sh) ---
It requires the literal 'failed to enable HDMI mode: %d at stage %s', extracts
the enum, and for every enumerator requires both a name in
sm750_mode_stage_names[] and an assignment of the form
'sdev->mode_stage = <STAGE>;' in the source. Negative controls:
  /mega/tmp/sm750/t27/drop-stage.c (assignment removed) -> exit 1,
    'GAP-B8: stage SM750_MODE_STAGE_TIMING is never assigned on the modeset path.'
  /mega/tmp/sm750/t27/drop-name.c (name removed) -> exit 1,
    'GAP-B8: stage SM750_MODE_STAGE_TIMING has no name in sm750_mode_stage_names.'

--- 8. what this does and does not close ---
It closes the identification gap only when a -110 recurs and dmesg names the
stage. No retry was added: a retry of a half-progressed modeset can corrupt
state, and the point of this branch is identification. GAP-B8 stays GAP.
